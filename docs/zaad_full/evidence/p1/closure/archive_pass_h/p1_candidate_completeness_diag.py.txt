"""Candidate-set completeness diagnostics (P1).

mode "exclude" (Stage 3/4, read-only): the existing stochastic class is generated at every NO_ACTION review
    (screen OPTIONAL set patched) but excluded from selection, so decisions are unchanged.
mode "escalate" (Stage 6/7): the request mode `no_action_escalation=True`; selection is live. At every review
    where the escalation fired, the stochastic candidate is re-generated by the ordinary screen path (OPTIONAL
    set patched, same request) and compared purchase-by-purchase: the bit-identity proof.
Writes closure/candidate_completeness_<mode>_<letters>.json."""
from __future__ import annotations
import json, sys
import zaad.full.planner as P
import zaad.full.replay as R
import zaad.full.screen as S
from zaad.full.contracts import Request
from zaad.full.model import Solution
from zaad.full.replay import replay
from zaad.full.settlement import PhysicalViolation
from zaad.full.valuation import common_candidate_value
from zaad.full.worlds import draw, forecast_all
from scripts.p1_fixtures import build, REVIEWS, SEEDS

pdf = lambda r: r.model_copy(update={k: Request.model_fields[k].default for k in ("stratified_scenarios", "timing_tiebreak", "analytic_shortage")})
FIX = {"P": "P_high_opening_stock", "C": "C_moq_four_weeks", "O": "O_low_opening_stock", "I": "I_max_pallets_ftl_cap",
       "A": "A_no_terms_good_incumbent", "G": "G_freight_tiers", "Q": "Q_volatile_demand_high_target", "T": "T_combined_realistic"}


def _traced_replay(req0, truth, patch_optional: bool, exclude: bool):
    orig_opt = dict(S.OPTIONAL); orig = P.plan
    traces, kept = [], []
    def traced(r):
        P.TRACE = {}
        rep = orig(r)
        t = dict(P.TRACE); P.TRACE = None
        t["orchestration"] = {k: rep["orchestration"].get(k) for k in ("escalated", "candidate_classes_requested", "milps_attempted")}
        t["final_p0"] = {a["source"]: a["quantity"] for a in rep["actions"] if a["action"] == "purchase" and a["period"] == 0}
        traces.append(t); return rep
    try:
        if patch_optional:
            S.OPTIONAL["NO_ACTION"] = ("stochastic_network",)
        P.EXCLUDE_FROM_SELECTION = frozenset({"stochastic_network"}) if exclude else frozenset()
        P.plan = traced; R.plan = traced
        out = replay(req0, truth, REVIEWS, "gated", keep_requests=kept)
    finally:
        P.plan = orig; R.plan = orig; P.EXCLUDE_FROM_SELECTION = frozenset()
        S.OPTIONAL.clear(); S.OPTIONAL.update(orig_opt)
    return out, traces, kept


def _screen_requested_stochastic(req):
    """The stochastic candidate as the ordinary screen path builds it (OPTIONAL patched, no escalation)."""
    orig_opt = dict(S.OPTIONAL)
    try:
        S.OPTIONAL["NO_ACTION"] = ("stochastic_network",)
        P.TRACE = {}
        P.plan(req.model_copy(update={"no_action_escalation": False}))
        t = dict(P.TRACE); P.TRACE = None
    finally:
        S.OPTIONAL.clear(); S.OPTIONAL.update(orig_opt)
    return t["candidates"].get("stochastic_network"), t


def _common(req, fits, c):
    purchases = {(k.rsplit(":", 1)[0], int(k.rsplit(":", 1)[1])): q for k, q in c["purchases"].items()}
    sol = Solution("OPTIMAL", purchases, {}, {}, {}, None, [], 0., 0, 0, None, 0., [])
    for po in req.pipeline:
        sol.amendments[(po.id, next(o.id for o in po.options if o.action == "keep"))] = po.remaining
    try:
        v = common_candidate_value(req, sol, fits); return v["total"], v["components"]
    except PhysicalViolation:
        return None, {}


def run(letter, seed, mode):
    req0 = pdf(build(FIX[letter], seed))
    if mode == "escalate":
        req0 = req0.model_copy(update={"no_action_escalation": True})
    truth = draw(req0, seed + 10000, 1)[0]
    out, traces, kept = _traced_replay(req0, truth, patch_optional=(mode == "exclude"), exclude=(mode == "exclude"))
    rows = []
    for i, (t, req) in enumerate(zip(traces, kept)):
        fits = forecast_all(req)
        cands = {}
        for name, c in t["candidates"].items():
            if c["status"] not in ("OPTIMAL", "FEASIBLE_TIME_LIMIT"):
                cands[name] = {"status": c["status"]}; continue
            total, comp = _common(req, fits, c)
            cands[name] = {"status": c["status"], "p0": round(sum(c["p0_purchases"].values()), 2), "sampled_cost": c["hedged_cost"], "exposure": c["exposure"],
                           "burden": c["burden"], "feasible_on_selection": c["feasible"], "service": c["service_stat"], "common": total,
                           "E_short": comp.get("expected_shortage_units"), "E_short_cost": comp.get("expected_shortage_cost"), "E_carry": comp.get("expected_carrying"),
                           "purchases": c["purchases"]}
        row = {"fixture": letter, "seed": seed, "review": i, "state": t["state"], "chosen": t["chosen"], "status": t["status"], "candidates": cands,
               "escalated": t["orchestration"]["escalated"], "final_p0": t["final_p0"], "gates": t.get("candidate_gates")}
        if mode == "escalate" and any(e.startswith("stochastic_network") for e in t["orchestration"]["escalated"]):
            ref, _ = _screen_requested_stochastic(req)
            mine = t["candidates"]["stochastic_network"]
            row["bit_identical_to_screen_requested"] = (ref is not None and ref["status"] == mine["status"] and ref["purchases"] == mine["purchases"])
        rows.append(row)
    return rows, out


if __name__ == "__main__":
    mode = sys.argv[1]; letters = sys.argv[2:] or ["P"]
    allrows = []
    for letter in letters:
        for seed in SEEDS:
            rows, out = run(letter, seed, mode); allrows += rows
            for r in rows:
                c = r["candidates"]; d, s = c.get("deterministic_network", {}), c.get("stochastic_network", {})
                esc = any(e.startswith("stochastic_network") for e in r["escalated"])
                if r["state"] != "NO_ACTION" and not esc: continue
                if "p0" not in s:
                    print(f"{letter} s{seed} r{r['review']:2d} {r['state'][:9]:9s} esc={int(esc)} stochastic {s.get('status')}"); continue
                fp0 = round(sum(r["final_p0"].values()), 1)
                print(f"{letter} s{seed} r{r['review']:2d} {r['state'][:9]:9s} esc={int(esc)} chosen {r['chosen'][:13]:13s} {r['status'][:6]} exec {fp0:6.1f} | p0 inc {c['incumbent'].get('p0',0):6.1f} det {d.get('p0',0):6.1f} stoch {s['p0']:6.1f} | sampled det {d.get('sampled_cost') and round(d['sampled_cost'])} stoch {round(s['sampled_cost']) if s['sampled_cost'] else None} | common det {d.get('common') and round(d['common'])} stoch {round(s['common']) if s['common'] else None} | E[short] det {d.get('E_short') and round(d['E_short'],1)} stoch {s['E_short'] and round(s['E_short'],1)} | svc det {d.get('service') and round(d['service'],3)} stoch {round(s['service'],3)}" + (f" | bit_identical {r.get('bit_identical_to_screen_requested')}" if mode == "escalate" and esc else ""))
            print(f"   {letter} seed {seed} replay cost ({mode}): {out['economic_cost']:.2f}; escalations {sum(1 for r in rows for e in r['escalated'] if e.startswith('stochastic_network'))}; states {dict((k, sum(1 for r in rows if r['state']==k)) for k in set(r['state'] for r in rows))}")
    json.dump(allrows, open(f"docs/zaad_full/evidence/p1/closure/candidate_completeness_{mode}_{''.join(letters)}.json", "w"), indent=1, default=float)
